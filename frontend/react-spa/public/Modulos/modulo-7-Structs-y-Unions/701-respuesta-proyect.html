void agregar_registro(int id, int tipo, float f_val, int i_val, const char *s_val, int e_critico, int warn, int calib) {
    if (contador_registros >= MAX_REGISTROS) {
        printf("ERROR: Log lleno. No se pudo agregar registro.\n");
        return;
    }

    // Acceso usando el operador [] para obtener la struct y . para acceder a los miembros
    Registro *nuevo_reg = &log_telemetria[contador_registros];

    // Rellenar miembros directos (Usando el operador flecha -> porque 'nuevo_reg' es un puntero)
    nuevo_reg->id_sensor = id;
    nuevo_reg->tipo_dato = tipo;
    nuevo_reg->timestamp = (long)time(NULL) + contador_registros * 60; // SimulaciÃ³n de tiempo

    // REQUISITO 3: Rellenar la UNION (Solo un miembro es vÃ¡lido)
    if (tipo == TIPO_FLOTANTE) {

    } else if (tipo == TIPO_ENTERO) {
        nuevo_reg->datos.valor_entero = i_val;
    } else if (tipo == TIPO_ESTADO) {
        // strncpy y asegurar terminador para evitar desbordamiento
        strncpy(nuevo_reg->datos.estado_str, s_val, 7);
        nuevo_reg->datos.estado_str[7] = '\0';
    }

    // REQUISITO 2: Rellenar los CAMPOS DE BITS
    // Acceso a la struct interna 'estado' (con .) y a sus miembros bit a bit (con .)
    nuevo_reg->estado.error_critico = e_critico;
    nuevo_reg->estado.warning = warn;
    nuevo_reg->estado.calibracion_ok = calib;
    nuevo_reg->estado.modo_baja_potencia = 0; // Por defecto
    nuevo_reg->estado.reservado = 0;

    contador_registros++;
}

            case TIPO_ENTERO:
                // Acceso a la union: 'reg->datos.valor_entero'

            case TIPO_ESTADO:
                // Acceso a la union: 'reg->datos.estado_str'

        }

        // Lectura de los CAMPOS DE BITS
        printf("  > Estado BITS: ");
