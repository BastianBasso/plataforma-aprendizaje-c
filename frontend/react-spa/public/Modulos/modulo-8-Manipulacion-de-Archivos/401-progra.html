    fclose(archivo_log);
}

    return EXIT_SUCCESS;

3. Lectura de un Ãšnico Registro con VerificaciÃ³n

#include <stdio.h>
#include <stdlib.h>

int main() {
    FILE *archivo = fopen("perfil.txt", "r");
    char nombre_leido[50];
    int edad_leida;
    float altura_leida;
    int items_leidos;
    
    if (archivo == NULL) return EXIT_FAILURE;
    
    // El formato debe coincidir EXACTAMENTE con el texto escrito
    items_leidos = fscanf(archivo, "Nombre: %[^,], Edad: %d, Altura: %f mts", 
                          nombre_leido, &edad_leida, &altura_leida);
    
    // Se esperan 3 elementos leÃ­dos (nombre, edad, altura)
    if (items_leidos == 3) { 
        printf("\nEjercicio 3: Lectura exitosa.\n");
        printf("Datos leÃ­dos: Nombre: %s | Edad: %d | Altura: %.2f mts\n", 
               nombre_leido, edad_leida, altura_leida);
    } else {
        printf("\nEjercicio 3: Fallo de lectura o formato incorrecto (solo se leyeron %d elementos).\n", items_leidos);
    }
    
    fclose(archivo);
    return EXIT_SUCCESS;
}

Nota: Se utiliza %[^,] en fscanf para leer la cadena nombre hasta la primera coma, ya que el nombre puede contener espacios.

10. Bucle de Lectura LÃ­nea por LÃ­nea (fgets para CSV)

#include <stdio.h>
#include <stdlib.h>

#define MAX_LINEA 100

int main() {
    // AsegÃºrese de que 'inventario.csv' existe con datos CSV

    char producto[30];

        contador++;
        // c. Usar sscanf para parsear la lÃ­nea (la coma es el delimitador)

        if (sscanf(buffer, "%[^,],%d,%f", producto, &cantidad, &precio) == 3) {
            printf("Registro %d | Producto: %-10s | Cantidad: %4d | Precio: %.2f\n", 
                   contador, producto, cantidad, precio);
        } else {
            printf("Error al parsear lÃ­nea %d.\n", contador);
        }
    }
    
    fclose(f);
    return EXIT_SUCCESS;
}
